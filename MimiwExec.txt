local function mkBtn(parent, text, x, y, w, h, color, cb)
    local b = Instance.new("TextButton")
    b.Size = UDim2.fromOffset(w, h) b.Position = UDim2.fromOffset(x, y)
    b.BackgroundColor3 = color or Color3.fromRGB(60, 30, 48) b.TextColor3 = Color3.new(1, 1, 1)
    b.Font = Enum.Font.GothamBold b.TextSize = 12 b.Text = text b.AutoButtonColor = true b.Parent = parent
    corner(b, 7)
    if cb then b.MouseButton1Click:Connect(cb) end

end

do
    local drag, st, og
    bar.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then

            win.Position = UDim2.new(og.X.Scale, og.X.Offset + d.X, og.Y.Scale, og.Y.Offset + d.Y)

local logN = 0
local function log(text, color)
    logN = logN + 1
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1, -8, 0, 0) l.AutomaticSize = Enum.AutomaticSize.Y l.BackgroundTransparency = 1
    l.TextColor3 = color or Color3.fromRGB(235, 220, 230) l.Font = Enum.Font.Code l.TextSize = 12
    l.TextXAlignment = Enum.TextXAlignment.Left l.TextWrapped = true l.Text = tostring(text) l.LayoutOrder = logN l.Parent = logSF
    if logN > 200 then
        for _, c in ipairs(logSF:GetChildren()) do
            if c:IsA("TextLabel") then c:Destroy() break end
        end
    end
    logSF.CanvasPosition = Vector2.new(0, 1e6)
end

local function run(src)
    src = tostring(src or "")
    if src:gsub("%s", "") == "" then log("Kotak kosong.", Color3.fromRGB(255, 210, 110)) return end
    if not loadstring then log("Executor ini tidak punya loadstring.", Color3.fromRGB(255, 110, 110)) return end
    local f, err = loadstring(src)
    if not f then log("Syntax error: " .. tostring(err), Color3.fromRGB(255, 110, 110)) showPage("Konsol") return end
    log("> menjalankan skrip...", PINK)
    task.spawn(function()
        local ok, e = pcall(f)
        if not ok then log("Error: " .. tostring(e), Color3.fromRGB(255, 110, 110)) else log("> selesai", Color3.fromRGB(140, 220, 160)) end
    end)
end

-- simpan & daftar
local listSF
local function ensureFolder()
    if makefolder and isfolder and not isfolder(FOLDER) then pcall(makefolder, FOLDER) end
end
local refresh
local bSave = mkBtn(pages.Editor, "Simpan", 0, 0, 70, 28, nil, function()
    if not writefile then log("Executor tidak punya writefile.", Color3.fromRGB(255, 210, 110)) return end
    local n = nameBox.Text:gsub("[^%w_%-%s]", ""):gsub("^%s+", ""):gsub("%s+$", "")
    if n == "" then log("Isi nama skrip dulu.", Color3.fromRGB(255, 210, 110)) return end
    ensureFolder()
    local ok, e = pcall(writefile, FOLDER .. "/" .. n .. ".lua", box.Text)
    if ok then log("Tersimpan: " .. n) refresh() else log("Gagal menyimpan: " .. tostring(e), Color3.fromRGB(255, 110, 110)) end
end)
bSave.Position = UDim2.new(0, 228, 1, row2)

listSF = Instance.new("ScrollingFrame")
listSF.Size = UDim2.new(1, 0, 1, -34) listSF.BackgroundColor3 = DARK listSF.BorderSizePixel = 0
listSF.ScrollBarThickness = 4 listSF.CanvasSize = UDim2.new() listSF.AutomaticCanvasSize = Enum.AutomaticSize.Y
listSF.Parent = pages.Skrip
corner(listSF, 8)
local lLay = Instance.new("UIListLayout") lLay.Padding = UDim.new(0, 4) lLay.Parent = listSF
local lPad = Instance.new("UIPadding") lPad.PaddingTop = UDim.new(0, 4) lPad.PaddingLeft = UDim.new(0, 4) lPad.PaddingRight = UDim.new(0, 4) lPad.Parent = listSF
refresh = function()
    for _, c in ipairs(listSF:GetChildren()) do if c:IsA("Frame") then c:Destroy() end end
    if not (listfiles and isfolder) then log("Daftar skrip butuh listfiles.", Color3.fromRGB(255, 210, 110)) return end
    ensureFolder()
    local ok, files = pcall(listfiles, FOLDER)
    if not ok then return end
    for _, path in ipairs(files) do
        local name = tostring(path):match("([^/\\]+)$")
        if name and (name:sub(-4) == ".lua" or name:sub(-4) == ".txt") then
            local r = Instance.new("Frame")
            r.Size = UDim2.new(1, -8, 0, 30) r.BackgroundTransparency = 1 r.Parent = listSF
            mkBtn(r, name, 0, 0, 0, 30, nil, function()
                local okr, src = pcall(readfile, path)
                if okr then box.Text = src nameBox.Text = name:gsub("%.%w+$", "") showPage("Editor") end
            end).Size = UDim2.new(1, -140, 1, 0)
            local rb = mkBtn(r, "Jalan", 0, 0, 56, 30, PINK, function()
                local okr, src = pcall(readfile, path)
                if okr then run(src) end
            end)
            rb.Position = UDim2.new(1, -132, 0, 0)
            local db = mkBtn(r, "Hapus", 0, 0, 56, 30, Color3.fromRGB(150, 40, 70), function()
                if delfile then pcall(delfile, path) refresh() end
            end)
            db.Position = UDim2.new(1, -70, 0, 0)
        end
    end
end
